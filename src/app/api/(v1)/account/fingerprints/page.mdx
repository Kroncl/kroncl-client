Методы для управления fingerprint-ключами — быстрый вход без пароля.

<MDXJsonSchemaBlock title='CREATE_REQUEST' fields={createFingerprintRequestFields} />

<MDXJsonSchemaBlock title='FINGERPRINT_WITH_KEY' fields={fingerprintWithKeyFields} />

<MDXStatusesBlock codes={createFingerprintResponseCodes} />
